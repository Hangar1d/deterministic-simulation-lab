cmake_minimum_required(VERSION 3.20)
project(deterministic_simulation_lab VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIMLAB_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(SIMLAB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(simlab_build_options INTERFACE)
if(MSVC)
    target_compile_options(simlab_build_options INTERFACE /W4 /permissive-)
    if(SIMLAB_WARNINGS_AS_ERRORS)
        target_compile_options(simlab_build_options INTERFACE /WX)
    endif()
else()
    target_compile_options(simlab_build_options INTERFACE
        -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow)
    if(SIMLAB_WARNINGS_AS_ERRORS)
        target_compile_options(simlab_build_options INTERFACE -Werror)
    endif()
endif()
if(SIMLAB_ENABLE_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizer mode requires a GCC/Clang toolchain supporting ASan and UBSan")
    endif()
    target_compile_options(simlab_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(simlab_build_options INTERFACE -fsanitize=address,undefined)
endif()

add_library(simlab_core src/simulation.cpp)
target_include_directories(simlab_core PUBLIC include)
target_link_libraries(simlab_core PRIVATE simlab_build_options)
add_library(simlab_codec src/codec.cpp)
target_link_libraries(simlab_codec PUBLIC simlab_core PRIVATE simlab_build_options)
add_executable(simlab app/main.cpp)
target_link_libraries(simlab PRIVATE simlab_codec simlab_build_options)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
